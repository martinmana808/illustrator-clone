# Pen Tool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** A full-parity Illustrator Pen tool: draw corner + smooth bezier anchors, live rubber-band preview, close paths, continue open paths, add/delete/convert anchors, and 45° angle constraint — all driven by a headless, unit-tested state machine.

**Architecture:** A `PenTool` class translates abstract pointer/keyboard events into Paper.js path mutations. It renders nothing itself, so it is fully testable in Node. A pure `anchors` helper module handles the segment/handle math (corner↔smooth conversion, mirrored handles, angle constraint). A thin React layer (`useActiveTool`) forwards real DOM pointer events into the active tool and draws overlay UI (preview, handles). Hit-testing uses Paper's `project.hitTest`.

**Tech Stack:** Paper.js, TypeScript, Vitest (jsdom + stubbed canvas, already configured).

## Global Constraints

- Paper.js only via `src/engine/*` and `src/tools/*`.
- Tools render nothing; they mutate the Paper document and expose state for the UI to read.
- All geometry helpers are pure and unit-tested.
- Hit tolerance constant: `HIT_TOLERANCE = 6` (project units), defined once in `src/tools/constants.ts`.
- Vector input type `Vec = { x: number; y: number }`; modifiers `Modifiers = { shift?; alt?; meta?; ctrl? }`.

```ts
import paper from "paper";
import type { Vec } from "@/tools/types";

const EPS = 1e-6;

export function isCorner(seg: paper.Segment): boolean {
  return seg.handleIn.length < 1e-4 && seg.handleOut.length < 1e-4;
}

export function isSmooth(seg: paper.Segment): boolean {
  if (seg.handleIn.length < 1e-4 || seg.handleOut.length < 1e-4) return false;
  // Collinear & opposite: angle between handleIn and -handleOut near 0.
  const inv = seg.handleOut.multiply(-1);
  return Math.abs(seg.handleIn.getAngle(inv)) < 1; // within 1 degree
}

export function makeCorner(seg: paper.Segment): void {
  seg.handleIn = new paper.Point(0, 0);
  seg.handleOut = new paper.Point(0, 0);
}

export function setSymmetricHandles(seg: paper.Segment, handleOut: Vec): void {
  seg.handleOut = new paper.Point(handleOut.x, handleOut.y);
  seg.handleIn = new paper.Point(-handleOut.x, -handleOut.y);
}

export function constrainAngle(v: Vec, stepDeg: number): Vec {
  const len = Math.hypot(v.x, v.y);
  if (len < EPS) return { x: 0, y: 0 };
  const angle = Math.atan2(v.y, v.x);
  const step = (stepDeg * Math.PI) / 180;
  const snapped = Math.round(angle / step) * step;
  return { x: Math.cos(snapped) * len, y: Math.sin(snapped) * len };
}
```

```ts
import paper from "paper";
import type { EditorDoc } from "@/engine/document";
import type { Vec, Modifiers, PenCursor } from "./types";

export class PenTool {
  private doc: EditorDoc;
  private path: paper.Path | null = null;
  private preview: Vec | null = null;

  constructor(doc: EditorDoc) {
    this.doc = doc;
  }

  get currentPath(): paper.Path | null {
    return this.path;
  }

  get previewPoint(): Vec | null {
    return this.preview;
  }

  private pt(p: Vec): paper.Point {
    return new this.doc.scope.Point(p.x, p.y);
  }

  pointerDown(p: Vec, _m: Modifiers = {}): void {
    this.doc.scope.activate();
    if (!this.path) {
      this.path = new this.doc.scope.Path();
      this.path.strokeColor = new this.doc.scope.Color(0, 0, 0);
      this.path.strokeWidth = 1;
    }
    this.path.add(this.pt(p));
  }

  pointerDrag(_p: Vec, _m: Modifiers = {}): void {}
  pointerUp(_p: Vec, _m: Modifiers = {}): void {}

  pointerMove(p: Vec, _m: Modifiers = {}): void {
    this.preview = this.path ? { x: p.x, y: p.y } : null;
  }

  finish(): void {
    this.path = null;
    this.preview = null;
  }

  hoverCursor(_p: Vec, _m: Modifiers = {}): PenCursor {
    return "pen";
  }
}
```

  private dragAnchor: paper.Segment | null = null;
  private downPoint: Vec | null = null;

  pointerDown(p: Vec, _m: Modifiers = {}): void {
    this.doc.scope.activate();
    if (!this.path) {

      this.path.strokeColor = new this.doc.scope.Color(0, 0, 0);
      this.path.strokeWidth = 1;

      dx = c.x;
      dy = c.y;
    }
    setSymmetricHandles(this.dragAnchor, { x: dx, y: dy });
  }

  pointerUp(_p: Vec, _m: Modifiers = {}): void {
    this.dragAnchor = null;
    this.downPoint = null;
  }

- [ ] **Step 1: Create the hook** `src/tools/useActiveTool.ts` — attaches a Paper `Tool`, forwards events into the active tool instance based on the store's `activeTool`, and draws the rubber-band preview + anchor overlays. Full implementation:

```ts
import paper from "paper";
import type { EditorDoc } from "@/engine/document";
import { PenTool } from "./pen";
import type { Modifiers, Vec } from "./types";
import { editorStore } from "@/state/store";

function mods(event: paper.ToolEvent): Modifiers {
  const k = (event.modifiers ?? {}) as Record<string, boolean>;
  return { shift: k.shift, alt: k.option || k.alt, meta: k.command || k.meta, ctrl: k.control };
}
function vec(pt: paper.Point): Vec {
  return { x: pt.x, y: pt.y };
}

export function installTools(doc: EditorDoc): () => void {
  const scope = doc.scope;
  scope.activate();
  const pen = new PenTool(doc);
  const tool = new scope.Tool();
  let overlay: paper.Path | null = null;

  function drawPreview() {
    if (overlay) { overlay.remove(); overlay = null; }
    if (editorStore.getState().activeTool !== "pen") return;
    const p = pen.previewPoint;
    const path = pen.currentPath;
    if (p && path && path.lastSegment) {
      overlay = new scope.Path({
        segments: [path.lastSegment.point, new scope.Point(p.x, p.y)],
        strokeColor: new scope.Color(0.4, 0.4, 0.9),
        dashArray: [4, 4],
      });
    }
    scope.view.update();
  }

  tool.onMouseDown = (e: paper.ToolEvent) => {
    if (editorStore.getState().activeTool !== "pen") return;
    pen.pointerDown(vec(e.point), mods(e));
    drawPreview();
  };
  tool.onMouseDrag = (e: paper.ToolEvent) => {
    if (editorStore.getState().activeTool !== "pen") return;
    pen.pointerDrag(vec(e.point), mods(e));
    scope.view.update();
  };
  tool.onMouseUp = (e: paper.ToolEvent) => {
    if (editorStore.getState().activeTool !== "pen") return;
    pen.pointerUp(vec(e.point), mods(e));
  };
  tool.onMouseMove = (e: paper.ToolEvent) => {
    if (editorStore.getState().activeTool !== "pen") return;
    pen.pointerMove(vec(e.point), mods(e));
    drawPreview();
  };
  tool.onKeyDown = (e: paper.KeyEvent) => {
    if (e.key === "enter" || e.key === "escape") { pen.finish(); drawPreview(); }
  };

  tool.activate();
  return () => { if (overlay) overlay.remove(); tool.remove(); };
}
```
